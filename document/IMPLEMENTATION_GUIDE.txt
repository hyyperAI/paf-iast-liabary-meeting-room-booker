Implementation:
const ProtectedRoute = ({ children }) => {
  const { user, userType } = useAuth();

  // Allow guests to access student pages
  if (userType === 'GUEST') {
    return <>{children}</>;
  }

  // Require auth for authenticated users
  if (!user) {
    return <Navigate to="/book" replace />;
  }

  return <>{children}</>;
};

const useAuthErrorHandler = () => {
  const { userType } = useAuth();

  const handleAuthError = (error) => {
    // Only handle errors for authenticated users
    if (userType !== 'GUEST' && error?.name === 'AuthError') {
      logout();
      navigate('/login');
      return true;
    }
    return false;
  };

  return { handleAuthError };
};

Name:

Phone Number:
- Required

- 11-15 digits

Email:
- Required
- Valid email format
- Student email preferred

Semester:
- Required
- Format: 1st, 2nd, 3rd, 4th, etc.
